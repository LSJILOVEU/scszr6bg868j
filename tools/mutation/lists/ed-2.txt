# ED-2: terrain brushes and their legalization (Content/MapTools/MapBrush.cs) and the facade (Presentation/Maps/MapDocument.Brush.cs).
# Filter: the ED-2 tests.
# Not listed: the falloff edge test d2 > r2 (the weight there is 0 either way); clamping the wanted height to the range (the
# interval is cut by the same range, so it cannot be seen); the bank band's c^2 <= B (weight 0 at B); Smooth reading the start
# heights (nothing changes before Legalize, so there is no updated value to read).

[ed2-weight-r4]
file = Assets/WarCore/Content/MapTools/MapBrush.cs
filter = FullyQualifiedName~Ed2
expect = WarCore.Tests.Ed2BrushTests
note = falloff divides by r^2 instead of r^4
original:
| return (int)(Full * k * k / (r2 * r2));
replacement:
| return (int)(Full * k * k / r2);

[ed2-raise-floor]
file = Assets/WarCore/Content/MapTools/MapBrush.cs
filter = FullyQualifiedName~Ed2
expect = WarCore.Tests.Ed2BrushTests
note = raise rounds toward negative infinity
original:
| work.Disc(i, j, radius, (h, w) => h + (long)amount * w / Full);
replacement:
| work.Disc(i, j, radius, (h, w) => h + FloorDiv((long)amount * w, Full));
