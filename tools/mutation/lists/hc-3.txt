# HC-3: one default unit speed (ContentPack.DefaultUnitSpeedMmPerSec), SpeedOfUnitType, the half-speed demos and the step cap
# checked when a recipe for a map with terrain or plates is built (MapRecipe.CheckStep). Filter: Hc3SpeedTests.
# The showcase house and terrain stages and the four standing demos load fixed layers and are not mutated here.

[hc3-named-ignored]
file = Assets/WarCore/Content/Data/ContentPack.cs
filter = FullyQualifiedName~Hc3SpeedTests
expect = WarCore.Tests.Hc3SpeedTests.S2_
note = SpeedOfUnitType ignores the type's own speed
original:
|             return unitTypeSpeeds[index] ?? DefaultUnitSpeedMmPerSec;
replacement:
|             return DefaultUnitSpeedMmPerSec;

[hc3-default-other]
file = Assets/WarCore/Content/Data/ContentPack.cs
filter = FullyQualifiedName~Hc3SpeedTests
expect = WarCore.Tests.Hc3SpeedTests.S2_
note = a type without a speed gets another number
original:
|             return unitTypeSpeeds[index] ?? DefaultUnitSpeedMmPerSec;
replacement:
|             return unitTypeSpeeds[index] ?? MaxUnitSpeedMmPerSec;

[hc3-unknown-default]
file = Assets/WarCore/Content/Data/ContentPack.cs
filter = FullyQualifiedName~Hc3SpeedTests
expect = WarCore.Tests.Hc3SpeedTests.S4_
note = an unknown unit type gets the default instead of an error
original:
|             if (index < 0) throw new ContentException("content pack '" + Id + "' has no unit type '" + unitTypeId + "'");
|             return unitTypeSpeeds[index] ?? DefaultUnitSpeedMmPerSec;
replacement:
|             if (index < 0) return DefaultUnitSpeedMmPerSec;
|             return unitTypeSpeeds[index] ?? DefaultUnitSpeedMmPerSec;
