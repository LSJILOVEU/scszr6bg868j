# Map spawning uses content only when explicitly enabled.

[gen-mapspawn-01-fixed-speed]
file = Assets/WarCore/Presentation/Bridge/MapRecipe.cs
filter = FullyQualifiedName~WarCore.Tests.GenMapSpawnTests
expect = WarCore.Tests.GenMapSpawnTests.EnabledSpawnsUseContentInTheWorld
note = Enabled map spawns keep the legacy speed
original:
| ? UnitSpawn.OfType(key, pack, type, squad, team, xMm, zMm, reserve, layer)
replacement:
| ? new UnitSpawn(key, pack.SpawnOfUnitType(type).Archetype, squad, xMm, zMm, SpeedMmPerSec, team, pack.WeaponOfUnitType(type), reserve, pack.MeleeOfUnitType(type), pack.SpawnOfUnitType(type).Block, layer)
