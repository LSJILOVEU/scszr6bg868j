# TRV-1a: the ramp prefab (MapPrefabs.Ramp) and door hints (MapGate.Hints). Filter: Trv1a tests.
# Not listed: the hint band's lower end made inclusive (gap >= StepRaw) is equivalent: a ground cell and a plate cell,
# both walkable, whose edge ends are both within StepRaw always have a seam, so the plate cell is reached and never named.

[trv1a-slope-strict]
file = Assets/WarCore/Content/MapTools/MapPrefabs.cs
filter = FullyQualifiedName~Trv1a
expect = WarCore.Tests.Trv1aRampTests.TheSlopeBoundIsInclusive
note = the slope bound is exclusive
original:
| if (den * CeilDiv(drop, cells) <= num * SimFix.Scale)
replacement:
| if (den * CeilDiv(drop, cells) < num * SimFix.Scale)
