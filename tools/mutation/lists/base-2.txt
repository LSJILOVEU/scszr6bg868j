# BASE-2: the generic hit resolver "damage" (EffectRaw of health scaled by falloff, no roll). Filter: damage, gen-hit and content error tests.

[dmg-no-falloff]
file = Assets/WarCore/Combat/DamageResolver.cs
filter = FullyQualifiedName~DamageResolver|FullyQualifiedName~GenHit|FullyQualifiedName~ContentErrorLock
expect = WarCore.Tests.DamageResolverTests.Damage_IsEffectScaledByFalloffWithoutRoll
note = falloff is ignored
original:
|             weapon.EffectRaw * falloffRaw / WarCore.Sim.SimFix.Scale;
replacement:
|             weapon.EffectRaw;

[dmg-kills]
file = Assets/WarCore/Combat/DamageResolver.cs
filter = FullyQualifiedName~DamageResolver|FullyQualifiedName~GenHit|FullyQualifiedName~ContentErrorLock
expect = WarCore.Tests.DamageResolverTests.Damage_WoundsUntilTheThirdHitKills
note = every hit kills, as under rwr.hit
original:
|             weapon.EffectRaw * falloffRaw / WarCore.Sim.SimFix.Scale;
replacement:
|             maxHealthRaw;

[dmg-rounds-up]
file = Assets/WarCore/Combat/DamageResolver.cs
filter = FullyQualifiedName~DamageResolver|FullyQualifiedName~GenHit|FullyQualifiedName~ContentErrorLock
expect = WarCore.Tests.DamageResolverTests.Damage_IsEffectScaledByFalloffWithoutRoll
note = rounding goes up
original:
|             weapon.EffectRaw * falloffRaw / WarCore.Sim.SimFix.Scale;
replacement:
|             (weapon.EffectRaw * falloffRaw + WarCore.Sim.SimFix.Scale - 1) / WarCore.Sim.SimFix.Scale;

[dmg-roll-read]
file = Assets/WarCore/Combat/DamageResolver.cs
filter = FullyQualifiedName~DamageResolver|FullyQualifiedName~GenHit|FullyQualifiedName~ContentErrorLock
expect = WarCore.Tests.DamageResolverTests.Damage_IsEffectScaledByFalloffWithoutRoll
note = the roll scales the damage
original:
|             weapon.EffectRaw * falloffRaw / WarCore.Sim.SimFix.Scale;
replacement:
|             weapon.EffectRaw * falloffRaw / WarCore.Sim.SimFix.Scale * (65536 - rollRaw) / 65536;

[dmg-not-allowed]
