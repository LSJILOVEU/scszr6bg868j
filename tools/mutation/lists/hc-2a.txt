# HC-2a: the see-through ring's threshold is the unit's body height at its stance (LegacySeeHeight without a body), its radius
# is 8/7 of the standing height (SeeRadius without a body), and the layer cut's headroom is WallSpan's walk clearance.
# Filter: Hc2aSeeTests (and Ml10 for the headroom). The hosts that call SeeRadiusFor are Unity code (no assembly here).

[hc2a-height-fixed]
file = Assets/WarCore/Presentation/Camera/LayerCut.cs
filter = FullyQualifiedName~Hc2a
expect = WarCore.Tests.Hc2aSeeTests.S2_TheThresholdIsTheBodysHeight
expect = WarCore.Tests.Hc2aSeeTests.S2_ThePlateThresholdIsTheBodysHeight
note = the threshold is always the legacy height
original:
| internal static float SeeHeight(BodySizes sizes, ViewEntity me) => sizes == null ? LegacySeeHeight : sizes.Top(me);
replacement:
| internal static float SeeHeight(BodySizes sizes, ViewEntity me) => LegacySeeHeight;

[hc2a-threshold-eye]
