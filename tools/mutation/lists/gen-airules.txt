# Explicit flat-world AI rules, recipe copying, and terrain rule precedence.

[gen-airules-01-flat-defaults]
file = Assets/WarCore/Presentation/Bridge/WorldRecipe.cs
filter = FullyQualifiedName~WarCore.Tests.GenAiRulesTests
expect = WarCore.Tests.GenAiRulesTests.FlatWithOption_ChasesOnFirstTick
note = flat worlds ignore the explicit AI pack
original:
|         internal AiRules AiRulesFor(bool terrain) => (terrain ? TerrainPack : AiPack)?.Section<AiRules>();
replacement:
|         internal AiRules AiRulesFor(bool terrain) => (terrain ? TerrainPack : null)?.Section<AiRules>();

[gen-airules-02-option-reversed]
file = Assets/WarCore/Presentation/Bridge/WorldRecipe.cs
filter = FullyQualifiedName~WarCore.Tests.GenAiRulesTests
expect = WarCore.Tests.GenAiRulesTests.FlatWithOption_ChasesOnFirstTick
expect = WarCore.Tests.GenAiRulesTests.FlatWithoutOption_KeepsOriginalState
note = flat AI is enabled only when the explicit option is absent
original:
|         internal AiRules AiRulesFor(bool terrain) => (terrain ? TerrainPack : AiPack)?.Section<AiRules>();
replacement:
|         internal AiRules AiRulesFor(bool terrain) => (terrain ? TerrainPack : AiPack == null ? TerrainPack : null)?.Section<AiRules>();

[gen-airules-03-terrain-source]
file = Assets/WarCore/Presentation/Bridge/WorldRecipe.cs
filter = FullyQualifiedName~WarCore.Tests.GenAiRulesTests
expect = WarCore.Tests.GenAiRulesTests.TerrainWithOption_KeepsTerrainRules
note = the explicit AI pack overrides the terrain world's existing rule source
original:
|         internal AiRules AiRulesFor(bool terrain) => (terrain ? TerrainPack : AiPack)?.Section<AiRules>();
replacement:
|         internal AiRules AiRulesFor(bool terrain) => (AiPack ?? TerrainPack)?.Section<AiRules>();
